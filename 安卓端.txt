这个错误表明Python找不到`mahjong_core`模块。让我们解决这个问题并完善项目结构。

## 解决方案

1. **确保文件结构正确**：
```
mahjong_calculator_v2/
├── mahjong_core.py    # 核心计算逻辑
├── mahjong_gui.py     # 图形界面
├── rules.json         # 番种规则
└── tiles/             # 麻将牌图片
```

2. **修改导入方式**：

在你的`mahjong_calculator_v2.py`中，尝试以下修改：

```python
# 修改前
from mahjong_core import MahjongFanCalculator

# 修改为
from .mahjong_core import MahjongFanCalculator
# 或者
try:
    from mahjong_core import MahjongFanCalculator
except ImportError:
    from .mahjong_core import MahjongFanCalculator
```

3. **或者使用完整项目结构**：

创建一个`__init__.py`文件使目录成为Python包：

```
mahjong_calculator_v2/
├── __init__.py
├── mahjong_core.py
├── mahjong_gui.py
├── rules.json
└── tiles/
```

然后使用相对导入。

由于模块导入问题，我建议将所有代码合并到一个文件中：

```python
# mahjong_calculator_v2.py
import json
import os
import sys
from collections import defaultdict
from typing import List, Dict, Set, Tuple
from PyQt5.QtWidgets import (QApplication, QMainWindow, QWidget, QVBoxLayout, QHBoxLayout, 
                            QPushButton, QLabel, QCheckBox, QSpinBox, QScrollArea)
from PyQt5.QtGui import QPixmap, QIcon
from PyQt5.QtCore import Qt, QSize

class MahjongFanCalculator:
    def __init__(self):
        # 直接嵌入rules.json内容
        self.rules = {
            "88番": [
                {"name": "大四喜", "fan": 88, "exclude": ["小四喜", "三风刻", "碰碰和", "圈风刻", "门风刻", "幺九刻"], "condition": "由4副风刻组成的和牌"},
                # 其他番种...
            ],
            # 其他番值...
        }
        self.reverse_index = self._build_reverse_index()
    
    def _build_reverse_index(self) -> Dict[str, List[Tuple[str, int]]]:
        index = defaultdict(list)
        for fan_value, fan_list in self.rules.items():
            for fan_info in fan_list:
                index[fan_info['name']].append((fan_value, fan_info))
        return dict(index)
    
    def calculate_fan(self, tiles: List[str], hand_config: Dict) -> Dict[str, int]:
        result = {}
        # 示例判断逻辑
        if len(tiles) == 14:
            result["示例番种"] = 1
        return result
    
    def is_winning_hand(self, tiles: List[str]) -> bool:
        return len(tiles) == 14  # 简化判断
    
    def get_possible_winning_tiles(self, tiles: List[str]) -> List[Tuple[str, Dict[str, int]]]:
        return [("W1", {"示例番种": 1})]  # 示例数据

class TileButton(QPushButton):
    def __init__(self, tile_type: str, parent=None):
        super().__init__(parent)
        self.tile_type = tile_type
        self.setFixedSize(44, 60)
        self.setIconSize(QSize(44, 60))
        self.setStyleSheet("QPushButton { border: none; }")
        
        # 使用资源系统或默认图标
        if os.path.exists(f"tiles/{tile_type}.png"):
            self.setIcon(QIcon(f"tiles/{tile_type}.png"))
        else:
            self.setText(tile_type[-1])
        
        self.count = 0
        self.max_count = 4 if tile_type[0] in ['W', 'B', 'T', 'F', 'J'] else 1
        self.clicked.connect(self.on_click)
    
    def on_click(self):
        self.count = (self.count + 1) % (self.max_count + 1)
        self.update_display()
        self.parent().on_tile_selected(self.tile_type)
    
    def update_display(self):
        if self.count == 0:
            self.setText(self.tile_type[-1])
        else:
            self.setText(f"{self.tile_type[-1]}({self.count})")

class MahjongCalculatorUI(QMainWindow):
    def __init__(self):
        super().__init__()
        self.calculator = MahjongFanCalculator()
        self.selected_tiles = []
        self.init_ui()
    
    def init_ui(self):
        self.setWindowTitle("国标麻将算番器")
        self.setMinimumSize(800, 600)
        
        main_widget = QWidget()
        self.main_layout = QVBoxLayout()
        
        # 创建麻将牌选择区域
        self.create_tile_selection()
        
        # 操作按钮
        self.create_action_buttons()
        
        # 特殊番种复选框
        self.create_special_fan_checkboxes()
        
        # 已选牌显示
        self.selected_tiles_label = QLabel("已选牌: ")
        self.main_layout.addWidget(self.selected_tiles_label)
        
        # 提示信息
        self.hint_label = QLabel("请选择14张牌")
        self.main_layout.addWidget(self.hint_label)
        
        # 结果展示
        self.result_label = QLabel("")
        self.main_layout.addWidget(self.result_label)
        
        main_widget.setLayout(self.main_layout)
        self.setCentralWidget(main_widget)
    
    def create_tile_selection(self):
        scroll = QScrollArea()
        scroll.setWidgetResizable(True)
        tile_widget = QWidget()
        tile_layout = QVBoxLayout()
        
        # 条子
        hbox = QHBoxLayout()
        for i in range(1, 10):
            btn = TileButton(f"T{i}", self)
            hbox.addWidget(btn)
        tile_layout.addLayout(hbox)
        
        # 筒子
        hbox = QHBoxLayout()
        for i in range(1, 10):
            btn = TileButton(f"B{i}", self)
            hbox.addWidget(btn)
        tile_layout.addLayout(hbox)
        
        # 万子
        hbox = QHBoxLayout()
        for i in range(1, 10):
            btn = TileButton(f"W{i}", self)
            hbox.addWidget(btn)
        tile_layout.addLayout(hbox)
        
        # 字牌
        hbox = QHBoxLayout()
        for i in range(1, 5):  # 东南西北
            btn = TileButton(f"F{i}", self)
            hbox.addWidget(btn)
        for i in range(1, 4):  # 中发白
            btn = TileButton(f"J{i}", self)
            hbox.addWidget(btn)
        tile_layout.addLayout(hbox)
        
        tile_widget.setLayout(tile_layout)
        scroll.setWidget(tile_widget)
        self.main_layout.addWidget(scroll)
    
    def create_action_buttons(self):
        hbox = QHBoxLayout()
        for action in ["立牌", "吃", "碰", "明杠", "暗杠", "重置"]:
            btn = QPushButton(action)
            btn.clicked.connect(lambda _, a=action: self.on_action_clicked(a))
            hbox.addWidget(btn)
        self.main_layout.addLayout(hbox)
    
    def create_special_fan_checkboxes(self):
        hbox = QHBoxLayout()
        for fan in ["自摸", "和绝张", "抢杠和", "海底捞月", "天和", "地和"]:
            cb = QCheckBox(fan)
            hbox.addWidget(cb)
        self.main_layout.addLayout(hbox)
    
    def on_tile_selected(self, tile_type):
        if len(self.selected_tiles) < 14:
            self.selected_tiles.append(tile_type)
        
        remaining = 14 - len(self.selected_tiles)
        if remaining > 0:
            self.hint_label.setText(f"请再选择{remaining}张牌")
        elif remaining == 0:
            self.hint_label.setText("已选择14张牌")
            if self.calculator.is_winning_hand(self.selected_tiles):
                self.result_label.setText("和牌! 正在计算番数...")
            else:
                self.result_label.setText("诈胡!")
        else:
            self.hint_label.setText("已超过14张牌")
        
        self.update_selected_tiles_display()
    
    def update_selected_tiles_display(self):
        tile_count = defaultdict(int)
        for tile in self.selected_tiles:
            tile_count[tile] += 1
        
        display_text = "已选牌: "
        for tile, count in sorted(tile_count.items()):
            display_text += f"{tile}({count}) "
        
        self.selected_tiles_label.setText(display_text)
    
    def on_action_clicked(self, action):
        if action == "重置":
            self.selected_tiles = []
            self.selected_tiles_label.setText("已选牌: ")
            self.hint_label.setText("请选择14张牌")
            self.result_label.setText("")

def main():

if __name__ == "__main__":
```
